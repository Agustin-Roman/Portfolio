Sub GuardadoDeDato()
    Dim wsForm As Worksheet, wsReg As Worksheet
    Dim tabla As ListObject
    Dim nuevaFila As ListRow

    ' Definir las hojas de trabajo
    Set wsForm = ThisWorkbook.Sheets("form2")
    Set wsReg = ThisWorkbook.Sheets("reg2")
    
    ' Obtener la tabla "Tabla2"
    Set tabla = wsReg.ListObjects("Tabla2")
    
    ' Agregar una nueva fila en la tabla
    Set nuevaFila = tabla.ListRows.Add

    ' Copiar y pegar los datos en la nueva fila
    With nuevaFila.Range
        .Cells(1, 1).Value = wsForm.Range("D7").Value
        .Cells(1, 2).Value = wsForm.Range("D9").Value
        .Cells(1, 3).Value = wsForm.Range("D11").Value
        .Cells(1, 4).Value = wsForm.Range("D13").Value
        .Cells(1, 5).Value = wsForm.Range("G13").Value
        .Cells(1, 6).Value = wsForm.Range("D15").Value
        .Cells(1, 7).Value = wsForm.Range("G15").Value
        .Cells(1, 8).Value = wsForm.Range("D17").Value
        .Cells(1, 9).Value = wsForm.Range("G17").Value
    End With

    ' Mensaje de confirmación
    MsgBox "Datos guardados.", vbInformation, "Confirmación"
End Sub

Sub BorrarForm()
'
' Borrar Formulario con Grabado
'
    Range("D7").Select
    Selection.ClearContents
    Range("D9").Select
    Selection.ClearContents
    Range("D11").Select
    Selection.ClearContents
    Range("D13").Select
    Selection.ClearContents
    Range("G13").Select
    Selection.ClearContents
    Range("G15").Select
    Selection.ClearContents
    Range("G17").Select
    Selection.ClearContents
    Range("D15").Select
    Selection.ClearContents
    Range("D17").Select
    Selection.ClearContents
    Range("D7").Select
End Sub

Sub BorrarContenidoFormulario()
    Dim respuesta As VbMsgBoxResult
    Dim ws As Worksheet
    Set ws = ThisWorkbook.Sheets("form2") ' Definir la hoja de trabajo
    
    ' Mostrar mensaje de confirmación
    respuesta = MsgBox("¿Está seguro de que quiere limpiar el formulario?", vbYesNo + vbQuestion, "Confirmar acción")
    
    ' Si la respuesta es sí, borrar el contenido de las celdas
    If respuesta = vbYes Then
        Dim c As Range
        Dim celdas As Variant
        celdas = Array("D7", "D9", "D11", "D13", "D15", "D17", "G13", "G15", "G17")
        
        ' Recorrer las celdas y borrar contenido incluso si están combinadas
        For Each c In ws.Range(Join(celdas, ","))
            c.MergeArea.ClearContents
        Next c
        
        MsgBox "Las celdas han sido borradas correctamente.", vbInformation, "Confirmación"
    Else
        MsgBox "No se realizó ninguna acción.", vbInformation, "Cancelado"
    End If
End Sub


Sub BorrarUltimoRegistro()
    Dim ws As Worksheet
    Dim tbl As ListObject
    Dim ultimaFila As Long

    ' Establecer la hoja y la tabla
    Set ws = ThisWorkbook.Sheets("reg2") ' Nombre de la hoja
    Set tbl = ws.ListObjects("Tabla2")  ' Nombre de la tabla

    ' Verificar si la tabla tiene filas de datos
    If tbl.ListRows.Count > 0 Then
        ' Obtener la última fila de la tabla
        ultimaFila = tbl.ListRows.Count
        
        ' Borrar el último registro de la tabla
        tbl.ListRows(ultimaFila).Delete
        
        MsgBox "Último registro eliminado correctamente.", vbInformation, "Operación Exitosa"
    Else
        MsgBox "No hay registros en la tabla.", vbExclamation, "Error"
    End If
End Sub
